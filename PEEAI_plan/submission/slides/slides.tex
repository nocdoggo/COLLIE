% 2. THE PROBLEM (27 s)
\begin{frame}{Twenty minutes before opening, the projector may be late}
  \begin{itemize}
    \item A \textbf{fictional campus showcase}: five activities, opens at 17:00, must finish by 18:00.
    \item At 16:40 a message: the projector \emphasize{may be about 25 minutes late}. Change too much and people lose time. Wait too long and the demos fail.
    \item AI agents now act on messages like this. \textbf{Acting on a wrong message} is the failure that matters.
  \end{itemize}
  \vspace{0.1cm}
  \begin{center}
    \includegraphics[width=0.86\textwidth]{fig_timeline.pdf}
  \end{center}

    end by six. At four forty, a message says the projector may be twenty five minutes late.
    Change too much and people lose time. Wait too long and the demos fail. AI agents now act on
    messages like this. Would you wait, check, or prepare a backup?
  }
\end{frame}

\begin{frame}{The coordinator may check, hold and make free changes, not spend}
  \begin{itemize}

    \item \textbf{Spending needs the organizer.} Participants and staff have their own narrow rights.
    \item Everything runs in a \textbf{fictional local sandbox}: no real bookings, purchases or messages.

    fictional local sandbox. [Skip if late: Participants and staff have their own narrow

\end{frame}

\begin{frame}{A message is a claim, not a fact}
  \begin{itemize}
    \item A \textbf{status check} asks the supplier's status service. It takes \emphasize{two simulated minutes} and returns a report, \textbf{not an arrival}.
  \end{itemize}
  \vspace{0.05cm}
  \begin{center}
    \includegraphics[height=7.3cm]{fig_claim.pdf}\\[2pt]

  \end{center}
  \note{
    \textbf{(35s, ends 1:47)} Here is the notice, from the delivery contact. Plan B treats it as
    a claim, not a fact. Before changing anything, it checks the one fact that could change the
    decision. A status check asks the supplier's status service. In this sandbox that takes two
    simulated minutes. The answer is delayed, with a reported arrival at five twenty five. Even
    that is a report, not an arrival, so the plan must still work if it slips.

% 5. COMMIT (48 s)
\begin{frame}{One checked change, one receipt}
  \begin{itemize}
    \item The \textbf{planner} builds the allowed repairs and checks each: \emphasize{16 built, 3 pass}. Best: a swap that \emphasize{costs \$0}.
    \item A \textbf{receipt} is the server's own record of an accepted change. The AI cannot write it.
  \end{itemize}
  \vspace{0.05cm}
  \begin{center}
    \includegraphics[width=0.84\textwidth]{fig_swap.pdf}\\[4pt]
    \fcolorbox{muted}{white}{\includegraphics[width=0.84\textwidth]{receipt_line.png}}\\[3pt]
    \figcaption{Shown: my fixed-rule baseline in an automated run. In a recorded practice run, the AI coordinator made the same check and the same swap at 16:43.}
  \end{center}
  \note{
    \textbf{(48s, ends 2:35)} Now the planner builds the allowed repairs from fixed templates,
    like a swap or a ten minute shift, and checks each against the deadline, the people, and the
    budget. Sixteen are built and three pass. The best is free: swap the demonstrations with the
    skills exchange, so the demos start at five thirty, after the reported arrival. It is
    applied at four forty three, and the server writes a receipt, its own record of the change.
    The AI cannot write it. These frames show my rule baseline. In a recorded run, the AI
    coordinator made the same check and the same swap.
  }
\end{frame}

\begin{frame}{Verify, prepare, commit}
  \vspace{-0.1cm}
  \begin{center}

  \end{center}
  \vspace{-0.1cm}

% 8. THE MODEL CHOOSES, THE SERVER CHECKS (30 s)
\begin{frame}{The model chooses, the server checks}
  \begin{itemize}
    \item A \textbf{policy} is the rule that picks the next step. Rules read only the messages they were written for. A \textbf{model} reads any wording, so it chooses and \emphasize{the server checks}.
  \end{itemize}
  \vspace{0.05cm}
  \begin{center}
    \includegraphics[width=0.9\textwidth]{fig_architecture.pdf}
  \end{center}
  \note{
    \textbf{(30s, ends 4:23)} A policy is the rule that picks the next step. P zero is fixed
    rules, which only read the messages they were written for. P one lets a model call the tools
    freely. P two, mine, lets the model pick from options the server built and checked. [Skip if
    late: P three is P two with every relevant check run first.] Every policy passes the same
    gate.
  }
\end{frame}

% 9. THE HELD-OUT TEST (50 s)
\begin{frame}{Held-out test: zero violations, no clear winner}
  \begin{itemize}
    \item \textbf{Held-out cases}: locked before the test and hidden from every policy. Code, prompts, model and limits were \textbf{frozen} first. A run is \textbf{met} when its final state and receipts show the case's outcome.
  \end{itemize}
  \begin{center}
    \includegraphics[width=\textwidth]{fig_results.pdf}
  \end{center}
  \vspace{-0.1cm}
  \begin{itemize}
    \item All \res{both:recorded} runs recorded. Every policy: \textbf{\res{both:viol} executed violations}, \textbf{\res{both:unnec} changes} on the 3 quiet cases. Refused: \res{campus:all:refused} campus and \res{pantry:all:refused} pantry requests (\res{both:stale} on a state that changed meanwhile).
    \item \emphasize{No consistent advantage for P2.} Campus model cost: P1 met \res{campus:P1:metk} of \res{campus:P1:metn} runs for \$\res{campus:P1:usd}, P3 \res{campus:P3:metk} for \$\res{campus:P3:usd}, P2 \res{campus:P2:metk} for \$\res{campus:P2:usd}. Fixed rules met every pantry run.
  \end{itemize}
  \note{
    \textbf{(50s, ends 5:13)} Everything so far came from development cases, practice cases I
    built with. Then I froze the code, the prompts, the model and the limits, and ran held-out
    cases, hidden from every policy. A first attempt hit the provider's rate limit, so I voided
    it and kept it on record. In the reported run all one hundred twenty runs were recorded. The
    server refused fifty five requests in all, and no policy executed a violation or changed a
    quiet event. But my P two showed no consistent advantage. P one, the model calling tools
    freely, met all twenty four campus runs, at three times P two's model cost. On the pantry,
    fixed rules met every run.
  }
\end{frame}
